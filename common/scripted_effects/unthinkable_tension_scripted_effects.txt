# Clamp tension variable between 0 and 100
unthinkable_clamp_tension = {
	if = {
		limit = {
			check_variable = { global.unthinkable_tension < 0 }
		}
		set_variable = { global.unthinkable_tension = 0 }
	}
	if = {
		limit = {
			check_variable = { global.unthinkable_tension > 100 }
		}
		set_variable = { global.unthinkable_tension = 100 }
	}
	# Check for 100% Tension - triggers immediate flashpoint if not already at war
	if = {
		limit = {
			check_variable = { global.unthinkable_tension > 99 }
			NOT = { has_war_with = SOV }
			NOT = { has_global_flag = unthinkable_super_event_unthinkable_done }
		}
		ENG = { country_event = { id = unthinkable.100 hours = 2 } }
	}
}

# Increase tension by 5%
unthinkable_add_tension_5 = {
	add_to_variable = { global.unthinkable_tension = 5 }
	unthinkable_clamp_tension = yes
}
